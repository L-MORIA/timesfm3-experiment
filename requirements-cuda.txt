#   uv venv .venv --python 3.12
#   uv pip install -r requirements-cuda.txt

# (stable PyTorch 2.7+ supports sm_120).
#
# After install, switch device="cpu" -> device="cuda" in load_kronos() and
# remove the os.environ["CUDA_VISIBLE_DEVICES"] = "-1" override at the top
# of the scripts (it forces TimesFM to CPU as well).
